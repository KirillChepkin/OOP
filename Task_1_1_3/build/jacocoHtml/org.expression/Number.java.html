package org.expression;

/**
 * Represents a constant double number.
 */
public class Number extends Expression {
    private final double number;

    public double getNumber() {
        return this.number;
    }

    public Number(double number) {
        this.number = number;
    }

    /**
     * The derivative of a constant number is always 0.
     */
    @Override
    public Expression derivative(String variable) {
        return new Number(0);
    }

    /**
     * Constant number is always evaluated to its value.
     *
     * @param assignment Must contain values for all variables inside the expression.
     *
     * @return Value of the expression.
     */
    @Override
    public double eval(Assignment assignment) {
        return this.number;
    }

    @Override
    public String toString() {
        return this.getClass().getSimpleName() + "(" + this.number + ")";
    }
}
